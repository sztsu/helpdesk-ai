\begin{frame}{Contextualização: O Gargalo no Suporte Técnico (ITSM)}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{O Cenário nos Departamentos de TI}
                \begin{itemize}
                    \item Alto volume de chamados de baixa complexidade (redefinição de senhas, dúvidas de rede, instalações).
                    \item Triagem manual consome tempo valioso de analistas sêniores.
                    \item Tempo de primeira resposta elevado para o usuário.
                \end{itemize}
            \end{block}
        \end{column}
        
        \begin{column}{0.48\textwidth}
            \begin{alertblock}{O Desafio de Desenvolvimento}
                \begin{itemize}
                    \item Desenvolver uma plataforma completa de Help Desk com backend tradicional (Node.js/Django, SQL, DevOps) exige alto custo e longo prazo de entrega.
                    \item Urgência de validação ágil com protótipos funcionais.
                \end{itemize}
            \end{alertblock}
        \end{column}
    \end{columns}
\end{frame}

% -----------------------------------------------------------------------------
% Slide 3: A Proposta: HelpDesk AI
% -----------------------------------------------------------------------------
\begin{frame}{A Solução: HelpDesk AI}
    \begin{block}{\textbf{Objetivo do Projeto}}
        Desenvolver um protótipo funcional de triagem automatizada e autoatendimento de incidentes técnicos de primeiro nível, integrando ferramentas No Code, automação Low Code e inteligência artificial generativa.
    \end{block}

    \vspace{0.3cm}
    \textbf{Diferenciais Operacionais:}
    \begin{itemize}
        \item \textbf{Triagem Imediata:} Classificação do incidente em segundos após a abertura.
        \item \textbf{Priorização Técnica Racional:} Redução do viés do usuário através de matriz P1 a P4 calculada por IA.
        \item \textbf{Autoatendimento Guiado:} O solicitante recebe um passo a passo para resolução autônoma antes da intervenção humana.
    \end{itemize}
\end{frame}

% -----------------------------------------------------------------------------
% Slide 4: O Tripe Metodologico
% -----------------------------------------------------------------------------
\begin{frame}{O Tripé Metodológico Aplicado}
    \begin{columns}[T]
        \begin{column}{0.32\textwidth}
            \begin{block}{\textbf{1. No Code}}
                \textbf{Google Forms \& Sheets}
                \begin{itemize}
                    \item Interface responsiva de abertura de chamados.
                    \item Base de dados tabular estruturada.
                    \item Zero linhas de código.
                \end{itemize}
            \end{block}
        \end{column}

        \begin{column}{0.32\textwidth}
            \begin{block}{\textbf{2. Low Code}}
                \textbf{Make (Integromat)}
                \begin{itemize}
                    \item Orquestração lógica orientada a eventos.
                    \item Mapeamento de payloads e variáveis.
                    \item Roteamento entre APIs.
                \end{itemize}
            \end{block}
        \end{column}

        \begin{column}{0.32\textwidth}
            \begin{block}{\textbf{3. Vibecode}}
                \textbf{Gemini / LLM}
                \begin{itemize}
                    \item Engenharia de prompt instrucional.
                    \item Classificação semântica de chamados.
                    \item Redação de roteiros de resolução.
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

% -----------------------------------------------------------------------------
% Slide 5: Camadas No Code e Low Code
% -----------------------------------------------------------------------------
\begin{frame}{Implementação: No Code e Low Code}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \textbf{No Code (Forms \& Sheets):}
            \begin{itemize}
                \item Coleta dados: Nome, E-mail, Título e Relato detalhado do problema.
                \item Gravação instantânea na planilha de controle do suporte.
            \end{itemize}
        \end{column}

        \begin{column}{0.48\textwidth}
            \textbf{Low Code (Make):}
            \begin{itemize}
                \item Monitora inserções de linhas na planilha.
                \item Encaminha o relato como payload para a API de IA.
                \item Atualiza a linha com Categoria e Prioridade geradas pela IA e dispara o e-mail via conector Gmail.
            \end{itemize}
        \end{column}
    \end{columns}

\begin{frame}{Camada Vibecode: A Lógica Cognitiva da IA}
    \textbf{Como o Vibecode opera na solução?}
    \begin{itemize}
        \item Substituição de lógicas condicionais complexas (if/else) por interpretação em linguagem natural.
        \item A IA recebe o texto livre do solicitante e gera uma saída estruturada em formato padronizado.
    \end{itemize}

    \vspace{0.2cm}
    \begin{exampleblock}{Diretrizes do Prompt de Triagem}
        \textit{"Você é um especialista em suporte técnico de TI (Help Desk). Analise o relato a seguir: \{\{Relato\}\}. Identifique a Categoria (Hardware, Software, Rede ou Acesso), atribua uma Prioridade técnica (P1 a P4) e elabore um roteiro de 3 a 4 passos práticos para o usuário tentar resolver a falha imediatamente."}
    \end{exampleblock}

    \vspace{0.2cm}
    \textbf{Resultado:} Tratamento humanizado, técnico e preciso para cada incidente relatado.
\end{frame}

% Slide 8: Reflexao Critica: 3 Vantagens
% -----------------------------------------------------------------------------
\begin{frame}{Reflexão Crítica: 3 Vantagens Principais}
    \begin{enumerate}
        \item \textbf{Redução Radical do Tempo de Resposta (SLA Primário):}
        \begin{itemize}
            \item O usuário obtém assistência técnica estruturada em segundos, viabilizando a resolução autônoma imediata.
        \end{itemize}
        \vspace{0.2cm}
        \item \textbf{Time-to-Market Acelerado e Custo de Infraestrutura Zero:}
        \begin{itemize}
            \item Solução funcional construída em poucas horas de trabalho com planos gratuitos (Google Workspace, Make e APIs de IA).
        \end{itemize}
        \vspace{0.2cm}
        \item \textbf{Compreensão Semântica e Flexibilidade (Vibecode):}
        \begin{itemize}
            \item A IA interpreta termos leigos e relatos imprecisos sem exigir regras rígidas de programação.
        \end{itemize}
    \end{enumerate}
\end{frame}

% -----------------------------------------------------------------------------
% Slide 9: Reflexao Critica: 3 Limitacoes
% -----------------------------------------------------------------------------
\begin{frame}{Reflexão Crítica: 3 Limitações e Desafios}
    \begin{enumerate}
        \item \textbf{Risco de Alucinação em Instruções Técnicas:}
        \begin{itemize}
            \item Modelos generativos podem recomendar comandos incorretos. A solução atua estritamente como triagem de suporte básico, não substituindo o técnico humano.
        \end{itemize}
        \vspace{0.2cm}
        \item \textbf{Gargalos de Cota e Latência em Planos Gratuitos:}
        \begin{itemize}
            \item O plano gratuito do Make impõe limites de operações mensais e verificação em intervalos de polling.
        \end{itemize}
        \vspace{0.2cm}
        \item \textbf{Governança e Privacidade de Dados (LGPD):}
        \begin{itemize}
            \item Informações internas trafegam por provedores de nuvem públicos. Em cenários corporativos sensíveis, exige-se contratação de instâncias dedicadas.
        \end{itemize}
    \end{enumerate}
\end{frame}
